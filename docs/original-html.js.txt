async function generate(){
  const err=document.getElementById("cerr"),h=CART.head;
  const hm=headMissing();markHead(hm);
  const miss=[...HEADF.filter(([k])=>String(h[k]||"").trim().length<10).map(z=>[z[0],z[1],z[2]]),...hm.map(z=>[z[0],z[1].replace(/ \(.*\)$/,""),z[1].match(/\((.*)\)$/)?.[1]||""])];
  document.querySelectorAll("[data-h]").forEach(el=>el.toggleAttribute("aria-invalid",miss.some(m=>m[0]===el.dataset.h)));
  if(miss.length){err.innerHTML=`<b>Preencha os dados da SC:</b> ${miss.map(m=>esc(m[1]+(m[2]?" ("+m[2]+")":""))).join(", ")}.${miss.some(m=>m[0]==="xobs1")?" O motivo precisa ter ao menos 10 caracteres.":""}`;err.hidden=false;const fx=document.querySelector("#headf [aria-invalid]");if(fx)fx.focus();return}
  err.hidden=true;
  if(!window.jspdf){toast("A biblioteca de PDF não carregou. Verifique a conexão e tente de novo.");return}
  const now=new Date(),pad=n=>String(n).padStart(2,"0");
  const at=`${pad(now.getDate())}/${pad(now.getMonth()+1)}/${now.getFullYear()} ${pad(now.getHours())}:${pad(now.getMinutes())}`;
  const num="SIM-"+now.getFullYear()+pad(now.getMonth()+1)+pad(now.getDate())+"-"+pad(now.getHours())+pad(now.getMinutes());
  if(await saveFile(`SC_${CART.f}_${num}.pdf`,buildPdf(num,at))){CART.done={num,at};saveCart();renderCart();toast("Relatório gerado. Nada foi enviado ao Protheus.")}
}
